    ```svelte
    <script lang="ts">
      import type { Gen2Command } from '$lib/protocol/gen2-types';

      let { command }: { command: Gen2Command } = $props();
      let expanded = $state(false);

      // Format key parameters per command type for display
      function formatParams(cmd: Gen2Command): string {
        switch (cmd.type) {
          case 'Query': {
            const p = cmd.params as import('$lib/protocol/gen2-types').QueryParams;
            return `DR=${p.dr} M=${p.m} Sess=${p.session} Q=${p.q} Tgt=${p.target}`;
          }
          case 'QueryRep': {
            const p = cmd.params as import('$lib/protocol/gen2-types').QueryRepParams;
            return `Sess=${p.session}`;
          }
          case 'QueryAdjust': {
            const p = cmd.params as import('$lib/protocol/gen2-types').QueryAdjustParams;
            return `Sess=${p.session} UpDn=${p.upDn}`;
          }
          case 'ACK': {
            const p = cmd.params as import('$lib/protocol/gen2-types').AckParams;
            return `RN16=0x${p.rn16.toString(16).padStart(4, '0').toUpperCase()}`;
          }
          case 'NAK': return '';
          case 'Select': {
            const p = cmd.params as import('$lib/protocol/gen2-types').SelectParams;
            return `Tgt=${p.target} Act=${p.action} Bank=${p.memBank} Len=${p.length}`;
          }
          case 'ReqRN': {
            const p = cmd.params as import('$lib/protocol/gen2-types').ReqRnParams;
            return `RN16=0x${p.rn16.toString(16).padStart(4, '0').toUpperCase()}`;
          }
          default: return '';
        }
      }

      function formatTimestamp(ms: number): string {
        return ms.toFixed(3) + ' ms';
      }

      function crcBadge(crcValid: boolean | null): { text: string; class: string } {
        if (crcValid === null) return { text: '', class: '' };
        return crcValid
          ? { text: 'CRC OK', class: 'text-green-400' }
          : { text: 'CRC FAIL', class: 'text-red-400' };
      }
    </script>

    <div class="border-b border-gray-800 px-3 py-1.5 hover:bg-gray-800/50 text-xs font-mono">
      <button
        onclick={() => (expanded = !expanded)}
        class="flex w-full items-center gap-3 text-left"
      >
        <!-- Timestamp -->
        <span class="w-20 shrink-0 text-gray-500">{formatTimestamp(command.timestamp)}</span>

        <!-- Command type -->
        <span class="w-20 shrink-0 font-semibold text-gray-200">{command.type}</span>

        <!-- Parameters -->
        <span class="flex-1 truncate text-gray-400">{formatParams(command)}</span>

        <!-- CRC badge -->
        {#if command.crcValid !== null}
          <span class="shrink-0 {crcBadge(command.crcValid).class}">{crcBadge(command.crcValid).text}</span>
        {/if}

        <!-- Expand arrow -->
        <svg
          class="h-3 w-3 shrink-0 text-gray-600 transition-transform {expanded ? 'rotate-90' : ''}"
          fill="none" stroke="currentColor" viewBox="0 0 24 24"
        >
          <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 5l7 7-7 7" />
        </svg>
      </button>

      <!-- Expanded: raw bits -->
      {#if expanded}
        <div class="mt-1 rounded bg-gray-900 px-2 py-1 text-[10px] text-gray-500 break-all">
          {command.rawBits.join('')}
        </div>
      {/if}
    </div>
    ```

    **2. Create `src/lib/components/DecodedOutputPanel.svelte`:**

    Main panel replacing the Timeline placeholder. Shows decoded commands with inventory round grouping and decode stats in the header.

    ```svelte
    <script lang="ts">
      import { decodeState } from '$lib/state/decode-state.svelte';
      import DecodedCommandRow from './DecodedCommandRow.svelte';

      // Group commands by roundId for visual separation
      // Use $derived for reactive grouping
      let groupedCommands = $derived(() => {
        // Build groups: array of { roundId, commands, complete }
        const groups: Array<{ roundId: number; commands: typeof decodeState.commands; complete: boolean }> = [];
        let currentGroup: typeof groups[0] | null = null;

        for (const cmd of decodeState.commands) {
          if (!currentGroup || currentGroup.roundId !== cmd.roundId) {
            currentGroup = { roundId: cmd.roundId, commands: [], complete: false };
            groups.push(currentGroup);
          }
          currentGroup.commands.push(cmd);
        }

        // Mark completeness from round updates
        for (const round of decodeState.rounds) {
          const group = groups.find(g => g.roundId === round.roundId);
          if (group) group.complete = round.complete;
        }

        return groups;
      });

      let crcPassRate = $derived(
        decodeState.stats.crcChecked > 0
          ? ((decodeState.stats.crcPassed / decodeState.stats.crcChecked) * 100).toFixed(1)
          : '---'
      );

      // Auto-scroll: scroll to bottom when new commands arrive
      let scrollContainer: HTMLDivElement;
      let autoScroll = $state(true);

      function handleScroll() {
        if (!scrollContainer) return;
        const atBottom = scrollContainer.scrollHeight - scrollContainer.scrollTop - scrollContainer.clientHeight < 50;
        autoScroll = atBottom;
      }

      $effect(() => {
        // Trigger on commands length change
        const _len = decodeState.commands.length;
        if (autoScroll && scrollContainer) {
          requestAnimationFrame(() => {
            scrollContainer.scrollTop = scrollContainer.scrollHeight;
          });
        }
      });
    </script>

    <div class="flex flex-col rounded border border-gray-700 bg-gray-900 min-h-[12rem] max-h-[24rem]">
      <!-- Header with stats -->
      <div class="flex items-center justify-between border-b border-gray-700 px-3 py-2">
        <span class="text-sm font-semibold text-gray-300">Decoded Commands</span>
        <div class="flex items-center gap-3 text-xs text-gray-500">
          <span>{decodeState.stats.totalCommands} cmds</span>
          <span>CRC: {crcPassRate}%</span>
          <span>{decodeState.stats.commandsPerSec.toFixed(0)} cmd/s</span>
        </div>
      </div>

      <!-- Command list -->
      <div
        bind:this={scrollContainer}
        onscroll={handleScroll}
        class="flex-1 overflow-y-auto"
      >
        {#if decodeState.commands.length === 0}
          <div class="flex h-full items-center justify-center text-sm text-gray-600 py-8">
            {#if decodeState.active}
              Waiting for RFID signals...
            {:else}
              Start capture to decode commands
            {/if}
          </div>
        {:else}
          {#each groupedCommands() as group (group.roundId)}
            <!-- Round separator -->
            {#if group.roundId > 0}
              <div class="flex items-center gap-2 bg-gray-950 px-3 py-0.5 text-[10px] text-gray-600">
                <span>Round {group.roundId}</span>
                <div class="h-px flex-1 bg-gray-800"></div>
                {#if decodeState.showBackscatter && !group.complete}
                  <span class="text-amber-500">incomplete</span>
                {/if}
              </div>
            {/if}

            <!-- Commands in this round -->
            {#each group.commands as command (command.sampleOffset)}
              <DecodedCommandRow {command} />
            {/each}
          {/each}

          <!-- Backscatter results (inline, when toggle is ON) -->
          {#if decodeState.showBackscatter}
            {#each decodeState.backscatter as result}
              <div class="flex items-center gap-2 border-b border-gray-800 px-3 py-1.5 text-xs font-mono bg-amber-950/20">
                <span class="text-amber-500" title="Experimental backscatter decode">&#9888;</span>
                <span class="text-gray-500">{result.encoding}</span>
                <span class="text-gray-400 flex-1 truncate">{result.bits.join('')}</span>
                <span class="text-amber-400">{(result.confidence * 100).toFixed(0)}%</span>
              </div>
            {/each}
          {/if}
        {/if}
      </div>
    </div>
    ```

    Design decisions (Claude's discretion per CONTEXT.md):
    - Console-style log format (monospace, compact rows) -- optimized for engineering verification, easy to scan many commands quickly
    - Inventory rounds separated by thin bar with round number and horizontal rule (not alternating backgrounds -- cleaner for dense data)
    - Incomplete round indicator only when backscatter toggle is ON (per locked decision in CONTEXT.md)
    - Backscatter results shown inline with warning triangle emoji and confidence percentage
    - Auto-scroll to bottom during live capture
